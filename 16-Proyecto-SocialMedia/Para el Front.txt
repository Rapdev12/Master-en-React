#  con pnpm
pnpm create vite red-social --template react-ts

# Estructura de mi Front
red-social/
├── src/
│   ├── components/
│   ├── pages/
│   ├── hooks/
│   ├── types/
│   │   └── index.ts       ← Aquí defines tus interfaces
│   ├── App.tsx            ← Nota la extensión .tsx
│   ├── main.tsx
│   └── vite-env.d.ts
├── tsconfig.json          ← Configuración de TypeScript
├── tsconfig.node.json
├── package.json
└── vite.config.ts
